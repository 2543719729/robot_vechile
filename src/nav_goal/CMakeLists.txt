cmake_minimum_required(VERSION 3.0.2)
project(nav_goal)
add_compile_options(-std=c++11)

# Text navigation and patrols can run without the proprietary voice SDK.
option(NAV_GOAL_WITH_AUDIO "Build text navigation with bobac3_audio TTS" OFF)
set(NAV_GOAL_COMPONENTS actionlib geometry_msgs move_base_msgs roscpp rospy tf)
find_package(catkin REQUIRED COMPONENTS ${NAV_GOAL_COMPONENTS})
if(NAV_GOAL_WITH_AUDIO)
  # Build/source the optional audio package first; it is not a public dependency.
  find_package(bobac3_audio REQUIRED)
endif()
catkin_package(CATKIN_DEPENDS ${NAV_GOAL_COMPONENTS})
include_directories(${catkin_INCLUDE_DIRS} ${bobac3_audio_INCLUDE_DIRS})

add_executable(nav_goal_node src/nav_goal.cpp)
add_executable(nav_multigoal_node src/nav_multigoal.cpp)
add_executable(text_nav_node src/text_nav.cpp)
if(NAV_GOAL_WITH_AUDIO)
  target_compile_definitions(text_nav_node PRIVATE NAV_GOAL_WITH_AUDIO)
  if(bobac3_audio_EXPORTED_TARGETS)
    add_dependencies(text_nav_node ${bobac3_audio_EXPORTED_TARGETS})
  endif()
endif()
foreach(target nav_goal_node nav_multigoal_node text_nav_node)
  target_link_libraries(${target} ${catkin_LIBRARIES})
  if(catkin_EXPORTED_TARGETS)
    add_dependencies(${target} ${catkin_EXPORTED_TARGETS})
  endif()
endforeach()
